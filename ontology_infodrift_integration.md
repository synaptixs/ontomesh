# Importing infodrift into ontology-toolkit — Phased Approach

**Author:** Neeraj Rohilla  
**Version:** 1.1 | April 2026  
**Host repo:** ontology-toolkit (this repo)  
**Dependency:** [infodrift (develop)](https://github.com/nrohilla-fibonacci/infodrift/tree/develop)

---

## Overview

This document describes a five-phase approach to importing the infodrift production ML drift monitoring library into the ontology-toolkit as a declared dependency. The ontology-toolkit is the primary, authoritative repo. infodrift is consumed as a library — its monitors, orchestrator, and reporters are driven by the toolkit's OWL class hierarchy, SHACL shapes, and JSON-LD context.

All integration code lives inside a new `toolkit/drift/` module within this repo. No changes are made to the infodrift repo itself.

```
Direction of control:   ontology-toolkit  →  consumes  →  infodrift
```

## Phase 2 — toolkit/drift/monitor.py — OWL entities drive infodrift monitors

**Effort:** ~1 day  
**What happens:** A new `toolkit/drift/monitor.py` reads OWL individuals already generated by the toolkit pipeline and registers each one as an entity in `DriftOrchestrator`. The toolkit's DB is the source of truth for baselines — no manual entity key configuration needed.

```python
# toolkit/drift/monitor.py

from rdflib import Graph, Namespace, RDF
from infodrift import DriftOrchestrator

FIVEG = Namespace("https://example.org/5g#")
ONTOLOGY_PATH = "ontology/enterprise.ttl"


def iri_to_entity_key(iri: str) -> str:
    """fiveg:AMF_v2_RegionA_N11  →  AMF_v2::RegionA::N11"""
    local = str(iri).replace(str(FIVEG), "")
    parts = local.split("_", maxsplit=2)
    return "::".join(parts)


class OntologyDriftMonitor:
    """Reads OWL individuals from the toolkit ontology and registers each
    as a monitored entity in infodrift's DriftOrchestrator."""

    def __init__(self, ontology_path: str = ONTOLOGY_PATH,
                 slo_config: dict | None = None):
        self.g = Graph().parse(ontology_path, format="turtle")
        self.orch = DriftOrchestrator(slo_config=slo_config or {
            "psi":         {"warning": 0.10, "critical": 0.20},
            "ttd_seconds": {"warning": 120,  "critical": 300},
            "fpr":         {"warning": 0.08, "critical": 0.15},
        })

    def register_all(self, baseline_features: dict,
                     baseline_logs: dict) -> list[str]:
        """Auto-register every OWL individual as an infodrift entity.

        Args:
            baseline_features: { entity_key: pd.DataFrame }
            baseline_logs:     { entity_key: list[str] }
        """
        registered = []
        for subj in self.g.subjects(RDF.type, None):
            if str(subj).startswith(str(FIVEG)):
                key = iri_to_entity_key(str(subj))
                if key in baseline_features:
                    self.orch.register_entity(
                        entity_key=key,
                        train_features_df=baseline_features[key],
                        numeric_features=["latency_ms", "error_rate"],
                        categorical_features=["status", "error_code"],
                        train_logs=baseline_logs.get(key, []),
                    )
                    registered.append(key)
        return registered

    def run(self, entity_key: str, prod_df, prod_logs=None,
            window_id="latest", **kwargs):
        return self.orch.run(
            entity_key=entity_key,
            prod_features_df=prod_df,
            prod_logs=prod_logs or [],
            window_id=window_id,
            **kwargs,
        )

    def orchestrator(self) -> DriftOrchestrator:
        return self.orch
```

**Usage:**

```python
from toolkit.drift.monitor import OntologyDriftMonitor

monitor = OntologyDriftMonitor(ontology_path="ontology/enterprise.ttl")
monitor.register_all(baseline_features=baselines, baseline_logs=logs)

alerts = monitor.run("AMF_v2::RegionA::N11",
                     prod_df=prod_df,
                     prod_logs=prod_logs,
                     window_id="2025-W30")
```

## Phase 3 — toolkit/drift/gate.py — SHACL shapes gate infodrift input

**Effort:** ~1 day  
**What happens:** `SHACLGate` validates any production DataFrame against `ontology/enterprise-shapes.ttl` (already generated by the toolkit pipeline) before passing it to any infodrift monitor. Bad data never reaches drift baselines.

```python
# toolkit/drift/gate.py

import pandas as pd
from rdflib import Graph, Namespace
from pyshacl import validate

FIVEG = Namespace("https://example.org/5g#")
SHAPES_PATH = "ontology/enterprise-shapes.ttl"


class SHACLGate:
    """Validates a production DataFrame against the toolkit's SHACL shapes
    before it reaches any infodrift monitor."""

    def __init__(self, shapes_path: str = SHAPES_PATH):
        self._shapes = Graph().parse(shapes_path, format="turtle")

    def validate(self, df: pd.DataFrame, entity_key: str) -> pd.DataFrame:
        """Return df unchanged if valid; raise ValueError on violation."""
        data_g = self._df_to_rdf(df, entity_key)
        conforms, _, report = validate(
            data_g, shacl_graph=self._shapes, abort_on_first=False
        )
        if not conforms:
            raise ValueError(
                f"[SHACLGate] {entity_key} failed validation:\n{report}"
            )
        return df

    def _df_to_rdf(self, df: pd.DataFrame, entity_key: str) -> Graph:
        g = Graph()
        g.bind("fiveg", FIVEG)
        key_clean = entity_key.replace("::", "_")
        for i, row in df.iterrows():
            subj = FIVEG[f"{key_clean}_{i}"]
            for col, val in row.items():
                if pd.notna(val):
                    g.add((subj, FIVEG[col], g.skolemize()))
        return g
```

**Usage:**

```python
from toolkit.drift.gate import SHACLGate

gate     = SHACLGate("ontology/enterprise-shapes.ttl")
clean_df = gate.validate(prod_df, "AMF_v2::RegionA::N11")
alerts   = monitor.run("AMF_v2::RegionA::N11", prod_df=clean_df)
```

## Phase 4 — toolkit/drift/enricher.py — Drift reports as toolkit ObservationRecords

**Effort:** ~2 days  
**What happens:** The raw JSON from infodrift's `HealthReporter.full_report()` is consumed by `DriftEnricher`, wrapped in the toolkit's JSON-LD `@context`, stamped with PROV-O triples, and stored in the toolkit's existing ObservationRecord store — making drift reports first-class toolkit artifacts.

```python
# toolkit/drift/enricher.py

import json
from datetime import datetime, timezone
from pathlib import Path

CONTEXT_PATH = "ontology/context.json"
DRIFT_NS     = "https://example.org/5g/drift#"
FIVEG_NS     = "https://example.org/5g#"


class DriftEnricher:
    """Wraps an infodrift health report in JSON-LD and PROV-O, then
    stores it as a toolkit ObservationRecord."""

    def __init__(self, context_path: str = CONTEXT_PATH,
                 obs_db_path: str = "output/obs_records.jsonl"):
        with open(context_path) as f:
            self._ctx = json.load(f).get("@context", {})
        self._db = Path(obs_db_path)
        self._db.parent.mkdir(parents=True, exist_ok=True)

    def enrich_and_store(self, raw_report: dict,
                         baseline_id: str, window_id: str) -> dict:
        key       = raw_report.get("entity_key", "")
        component = key.split("_")[0] if key else "Unknown"
        enriched  = {
            "@context":              self._ctx,
            "@id":                   FIVEG_NS + key.replace("::", "_"),
            "@type":                 FIVEG_NS + component,
            "drift:windowId":        window_id,
            "prov:wasGeneratedBy":   DRIFT_NS + f"MonitoringRun_{window_id}",
            "prov:wasAttributedTo":  FIVEG_NS + key.replace("::", "_"),
            "prov:hadPrimarySource": DRIFT_NS + f"Baseline_{baseline_id}",
            "prov:atTime":           datetime.now(timezone.utc).isoformat(),
            **{self._map(k): v for k, v in raw_report.items()
               if k != "entity_key"},
        }
        with self._db.open("a") as f:
            f.write(json.dumps(enriched) + "\n")
        return enriched

    @staticmethod
    def _map(key: str) -> str:
        return {
            "psi_score":    "drift:psiScore",
            "drift_level":  "drift:driftLevel",
            "js_score":     "drift:jsScore",
            "brier_score":  "drift:brierScore",
            "ece":          "drift:ece",
            "tpr":          "drift:tpr",
            "fpr":          "drift:fpr",
            "median_ttd_s": "drift:medianTTDSeconds",
        }.get(key, key)
```

**Usage:**

```python
from infodrift import HealthReporter
from toolkit.drift.enricher import DriftEnricher

reporter = HealthReporter(monitor.orchestrator())
enricher = DriftEnricher("ontology/context.json", "output/obs_records.jsonl")

raw = json.loads(reporter.full_report(as_json=True))
obs = enricher.enrich_and_store(raw, baseline_id="2024W01",
                                     window_id="2025-W30")
```

## Phase 5 — toolkit/drift/propagator.py — RuntimeClient becomes drift-aware

**Effort:** ~1 week  
**What happens:** `OWLPropagator` consumes infodrift alerts and traverses the OWL class hierarchy to escalate monitoring on dependent and sibling NFs. `RuntimeClient.ask()` is updated to inject active drift alerts into the LLM prompt automatically, making every question drift-aware.

```python
# toolkit/drift/propagator.py

from rdflib import Graph, Namespace
from rdflib.namespace import RDFS

FIVEG         = Namespace("https://example.org/5g#")
ONTOLOGY_PATH = "ontology/enterprise.ttl"


class OWLPropagator:
    """Traverses the OWL class graph to escalate monitoring on NFs
    that depend on or are siblings of an alerting entity."""

    def __init__(self, ontology_path: str = ONTOLOGY_PATH):
        self.g = Graph().parse(ontology_path, format="turtle")

    def dependents(self, entity_key: str) -> list[str]:
        component = entity_key.split("_")[0]
        owl_class = FIVEG[component]
        related: set[str] = set()

        for _, _, obj in self.g.triples((owl_class, None, None)):
            local = str(obj).replace(str(FIVEG), "")
            if local and local != component:
                related.add(local)

        for parent in self.g.objects(owl_class, RDFS.subClassOf):
            for sibling in self.g.subjects(RDFS.subClassOf, parent):
                local = str(sibling).replace(str(FIVEG), "")
                if local != component:
                    related.add(local)

        suffix = "_".join(entity_key.split("_")[1:])
        return [f"{cls}_{suffix}" for cls in related]

    def propagate(self, alert: dict, monitor) -> list[str]:
        escalated = []
        for dep_key in self.dependents(alert.get("entity_key", "")):
            try:
                monitor.orchestrator().increase_monitoring_frequency(dep_key)
                escalated.append(dep_key)
            except (KeyError, AttributeError):
                pass
        return escalated
```

**RuntimeClient update (`runtime/client.py` excerpt):**

```python
from toolkit.drift.propagator import OWLPropagator

class RuntimeClient:
    def __init__(self, ..., drift_monitor=None):
        ...
        self._drift_monitor  = drift_monitor
        self._owl_propagator = OWLPropagator() if drift_monitor else None

    def ask(self, question: str, flavor: str = "default") -> dict:
        drift_context = ""
        if self._drift_monitor:
            recent = self._drift_monitor.orchestrator().recent_alerts(n=5)
            if recent:
                drift_context = "\n\nActive drift alerts:\n" + \
                    "\n".join(f"- {a['entity_key']}: {a['message']}"
                              for a in recent)
                for a in recent:
                    self._owl_propagator.propagate(a, self._drift_monitor)

        payload = self._assembler.build(question + drift_context, flavor)
        ...
```

## End-to-end wiring

```python
# main_with_drift.py  — complete example inside ontology-toolkit

import json
from toolkit.drift.monitor    import OntologyDriftMonitor
from toolkit.drift.gate       import SHACLGate
from toolkit.drift.enricher   import DriftEnricher
from toolkit.drift.propagator import OWLPropagator
from infodrift                import HealthReporter
from runtime.client           import RuntimeClient

gate     = SHACLGate("ontology/enterprise-shapes.ttl")
monitor  = OntologyDriftMonitor("ontology/enterprise.ttl")
monitor.register_all(baseline_features=baselines, baseline_logs=logs)

enricher   = DriftEnricher("ontology/context.json", "output/obs_records.jsonl")
propagator = OWLPropagator("ontology/enterprise.ttl")
client     = RuntimeClient(..., drift_monitor=monitor)

entity_key = "AMF_v2::RegionA::N11"
clean_df   = gate.validate(prod_df, entity_key)          # P3 — SHACL gate
alerts     = monitor.run(entity_key,                     # P2 — infodrift run
                         prod_df=clean_df,
                         prod_logs=prod_logs,
                         window_id="2025-W30")

reporter   = HealthReporter(monitor.orchestrator())
raw_report = json.loads(reporter.full_report(as_json=True))
obs_record = enricher.enrich_and_store(                  # P4 — ObsRecord
    raw_report, baseline_id="2024W01", window_id="2025-W30"
)

for alert in alerts:                                     # P5 — propagate
    if alert.get("severity") in ("warning", "critical"):
        escalated = propagator.propagate(alert, monitor)
        print(f"Escalated: {escalated}")

result = client.ask("Which NFs have active drift alerts?", flavor="fiveg")
print(result["answer"])
```
